% ── \keystroke: dibuja una "tecla" inline (ej. Ctrl+S) ──
% baseline=(key.base) alinea verticalmente la tecla con el texto circundante.
% inner xsep/ysep controlan el padding horizontal/vertical dentro de la tecla.
% rounded corners da las esquinas redondeadas estilo keycap.
\newcommand{\keystroke}[1]{%
  \tikz[baseline=(key.base)]{%
    \node[fill=lightgray,          % fondo gris claro (simula tecla física)
          draw=medgray,             % borde gris medio
          rounded corners=3pt,      % esquinas redondeadas (radio 3pt)
          inner xsep=3pt,           % padding horizontal
          inner ysep=2pt,           % padding vertical
          font=\bfseries\ttfamily\scriptsize  % negrita + monoespaciada + pequeña
    ] (key) {#1};%
  }%
}

\begin{center}
% ═══════════════════════════════════════════════════════════════
% DIAGRAMA DE FLUJO — Decisión Overleaf → VS Code
%
% Estructura planificada:
%   [Overleaf] → ◇¿No compila?◇ → No: [Seguir]  |  Sí: [VS Code]
%                                                       ↓
%                                             [¡Quédate en VS Code!]
%
% Ancho de columna disponible ≈ 8.7 cm (landscape A4, 3 columnas).
% Las cajas usan text width=3.6cm con xshift=±2.3cm → total ≈ 8.2cm.
% ═══════════════════════════════════════════════════════════════
\begin{tikzpicture}[
  % ── Separación vertical por defecto entre nodos consecutivos ──
  node distance=0.7cm,
  % ── Fuente base para todo nodo que no defina la suya ──
  every node/.style={font=\small},
  % ── Estilo "box": rectángulo azul redondeado (pasos del flujo) ──
  box/.style={
    rectangle,
    rounded corners=5pt,      % esquinas redondeadas
    draw=accentblue,           % borde azul
    fill=accentblue!8,         % relleno azul 8% (casi blanco)
    text width=3.6cm,          % ancho fijo del texto dentro de la caja
    align=center,              % texto centrado
    minimum height=0.7cm,      % altura mínima
    font=\small
  },
  % ── Estilo "decision": rombo rojo (pregunta sí/no) ──
  decision/.style={
    diamond,
    aspect=2.5,                % ensancha el rombo (ancho = 2.5× alto)
                               %   sin esto el rombo es muy puntiagudo
    draw=accentred,
    fill=accentred!8,
    text width=2.2cm,          % ancho del texto dentro del rombo
    align=center,
    font=\small\bfseries,
    inner sep=3pt              % margen interno (respiro alrededor del texto)
  },
  % ── Estilo "arrow": flecha gris con punta stealth ──
  arrow/.style={->, thick, >=stealth, color=medgray},
  % ── Estilo para la flecha de la rama recomendada (Sí → verde) ──
  greenarrow/.style={->, thick, >=stealth, color=accentgreen}
]

  % ── Nodo 1: punto de partida ──
  \node[box] (work) {Trabajar en \textbf{Overleaf}};

  % ── Nodo 2: pregunta de decisión ──
  %    "below=of work" lo coloca 0.7cm debajo (node distance)
  \node[decision, below=of work] (broke) {¿No compila?};

  % ── Nodo 3 (rama izquierda): camino "No" ──
  %    below=1cm of broke → 1cm debajo del rombo
  %    xshift=-2.3cm       → desplazado 2.3cm a la IZQUIERDA
  \node[box, below=1cm of broke, xshift=-2.3cm] (no)
    {Seguir en Overleaf};

  % ── Nodo 4 (rama derecha): camino "Sí" — acción de rescate ──
  %    xshift=+2.3cm → desplazado 2.3cm a la DERECHA
  \node[box, below=1cm of broke, xshift=2.3cm] (yes)
    {Compilar en VS Code};

  % ── Nodo 5: conclusión verde (destino final de la ruta recomendada) ──
  %    fill/draw en verde para resaltar visualmente como meta
  %    line width=1.5pt → borde más grueso que las cajas azules
  \node[box, below=of yes,
        fill=accentgreen!15, draw=accentgreen,
        line width=1.5pt,
        font=\small\bfseries       % negrita para enfatizar la conclusión
  ] (fix) {¡Quédate en VS Code!};

  % ── Flechas ──
  \draw[arrow]      (work)  -- (broke);                                           % inicio → decisión
  \draw[arrow]      (broke) -- node[left,  font=\scriptsize\color{medgray}]  {No}  (no);   % rama izq (gris, camino pasivo)
  \draw[greenarrow] (broke) -- node[right, font=\scriptsize\bfseries] {Sí}         (yes);  % rama der (verde, camino recomendado)
  \draw[greenarrow] (yes)   -- (fix);                                              % rescate → quedarse

\end{tikzpicture}
\end{center}

\sectiontitle{\faLifeRing}{Overleaf no compila → VS Code}

{\small\color{medgray} No es tu culpa. Overleaf a veces no procesa ciertos paquetes. VS Code sí.}

\vspace{4pt}

\begin{enumerate}[leftmargin=*, itemsep=3pt, font=\bfseries\color{accentblue}]
  \item \textbf{Descargar} de Overleaf:\\
    {\small Menu → Download → \textbf{Source} (.zip)}
  \item \textbf{Descomprimir} el .zip:\\
    {\small Win: clic derecho → Extraer todo\\
    Mac: doble clic}
  \item \textbf{Abrir} en VS Code:\\
    {\small File → \textbf{Open Folder} → la carpeta}\\
    {\small\color{accentred} ¡La carpeta, NO el archivo!}\\
    {\scriptsize Si pregunta ``Trust authors?'' \textrightarrow~\textbf{Yes}}
  \item \textbf{Compilar}: atajo de Build\\
    {\small Primera vez tarda 1--3 min (normal)}\\
    {\scriptsize Si MiKTeX pide instalar paquetes \textrightarrow~\textbf{Install}/\textbf{Yes}}
  \item \textbf{Ver PDF}: atajo de Ver PDF
  \item \textbf{¡Listo!} Quédate en VS Code\\
    {\small Es gratis, rápido y más estable.\\
    Tu PDF está en la misma carpeta.}
\end{enumerate}

\vspace{6pt}

\warningbox{El PDF SOLO se ve dentro de VS Code.}

\vspace{4pt}

\noindent{\small\color{darktext} Si el PDF está abierto en \textbf{WPS Office}, \textbf{Adobe} u otro programa, VS Code no puede actualizarlo. Cierra el PDF en el otro programa primero.}
